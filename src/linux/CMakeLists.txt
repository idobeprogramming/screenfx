find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Widgets DBus OpenGL)
set(CMAKE_AUTOMOC ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
include(GNUInstallDirs)

add_library(screenfx_linux_core STATIC ShaderSource.cpp LinuxSettings.cpp)
target_include_directories(screenfx_linux_core PUBLIC "${PROJECT_SOURCE_DIR}/src")
target_link_libraries(screenfx_linux_core PUBLIC Qt6::Core)
target_compile_options(screenfx_linux_core PRIVATE -Wall -Wextra -Wpedantic)

add_library(screenfx_linux_backend STATIC CompositorBackend.cpp CompositorBackend.h)
target_include_directories(screenfx_linux_backend PUBLIC "${PROJECT_SOURCE_DIR}/src")
target_link_libraries(screenfx_linux_backend PUBLIC screenfx_linux_core Qt6::DBus)
target_compile_options(screenfx_linux_backend PRIVATE -Wall -Wextra -Wpedantic)
add_executable(screenfx main.cpp)
target_compile_definitions(screenfx PRIVATE SCREENFX_VERSION="${PROJECT_VERSION}")
target_link_libraries(screenfx PRIVATE screenfx_linux_backend Qt6::Widgets)
target_compile_options(screenfx PRIVATE -Wall -Wextra -Wpedantic)
set_target_properties(screenfx PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
install(TARGETS screenfx RUNTIME DESTINATION "${CMAKE_INSTALL_BINDIR}")
install(FILES "${PROJECT_SOURCE_DIR}/packaging/screenfx.desktop" DESTINATION "${CMAKE_INSTALL_DATADIR}/applications")

option(SCREENFX_BUILD_KWIN "Build the native KWin 6.6 effect (disable for Hyprland-only builds)" ON)
if(SCREENFX_BUILD_KWIN)
    find_package(ECM 6.0 REQUIRED NO_MODULE)
    list(APPEND CMAKE_MODULE_PATH ${ECM_MODULE_PATH})
    find_package(KWin 6.6 REQUIRED)
    find_package(KF6GlobalAccel REQUIRED)
    include(KDEInstallDirs6)
    add_library(screenfx_kwin MODULE kwin/ScreenFxEffect.cpp)
    target_link_libraries(screenfx_kwin PRIVATE KWin::kwin KF6::GlobalAccel Qt6::DBus)
    target_compile_options(screenfx_kwin PRIVATE -Wall -Wextra -Wpedantic)
    set_target_properties(screenfx_kwin PROPERTIES PREFIX "" OUTPUT_NAME "screenfx"
        LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/kwin/effects/plugins")
    install(TARGETS screenfx_kwin LIBRARY DESTINATION "${KDE_INSTALL_PLUGINDIR}/kwin/effects/plugins")
    if(SCREENFX_BUILD_TESTS)
        add_executable(screenfx_linux_desktop_tests "${PROJECT_SOURCE_DIR}/tests/LinuxDesktopTests.cpp")
        target_link_libraries(screenfx_linux_desktop_tests PRIVATE screenfx_linux_backend Qt6::Widgets)
        add_dependencies(screenfx_linux_desktop_tests screenfx_kwin)
    endif()
endif()

if(SCREENFX_BUILD_TESTS)
    enable_testing()
    add_executable(screenfx_linux_shader_tests "${PROJECT_SOURCE_DIR}/tests/LinuxShaderTests.cpp")
    target_link_libraries(screenfx_linux_shader_tests PRIVATE screenfx_linux_core Qt6::Gui Qt6::OpenGL)
    add_executable(screenfx_linux_settings_tests "${PROJECT_SOURCE_DIR}/tests/LinuxSettingsTests.cpp")
    target_link_libraries(screenfx_linux_settings_tests PRIVATE screenfx_linux_core)
    add_executable(screenfx_linux_backend_tests "${PROJECT_SOURCE_DIR}/tests/LinuxBackendTests.cpp")
    target_link_libraries(screenfx_linux_backend_tests PRIVATE screenfx_linux_backend)
    add_dependencies(screenfx_linux_backend_tests screenfx)
    add_executable(screenfx_linux_panel_tests "${PROJECT_SOURCE_DIR}/tests/LinuxPanelTests.cpp")
    set_source_files_properties("${PROJECT_SOURCE_DIR}/tests/LinuxPanelTests.cpp" PROPERTIES SKIP_AUTOMOC ON)
    target_include_directories(screenfx_linux_panel_tests PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/screenfx_autogen/include")
    target_compile_definitions(screenfx_linux_panel_tests PRIVATE SCREENFX_VERSION="${PROJECT_VERSION}")
    target_link_libraries(screenfx_linux_panel_tests PRIVATE screenfx_linux_backend Qt6::Widgets)
    add_dependencies(screenfx_linux_panel_tests screenfx)
    add_test(NAME screenfx_linux_settings COMMAND screenfx_linux_settings_tests)
    find_program(SCREENFX_DBUS_RUN_SESSION dbus-run-session REQUIRED)
    add_test(NAME screenfx_linux_panel COMMAND "${SCREENFX_DBUS_RUN_SESSION}" -- $<TARGET_FILE:screenfx_linux_panel_tests>)
    set_tests_properties(screenfx_linux_panel PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 30)
    add_test(NAME screenfx_linux_backend COMMAND "${SCREENFX_DBUS_RUN_SESSION}" -- $<TARGET_FILE:screenfx_linux_backend_tests> $<TARGET_FILE:screenfx>)
    find_program(SCREENFX_XVFB_RUN xvfb-run)
    if(SCREENFX_XVFB_RUN)
        add_test(NAME screenfx_linux_shaders COMMAND sh "${PROJECT_SOURCE_DIR}/tests/with-xvfb.sh" $<TARGET_FILE:screenfx_linux_shader_tests>)
        set_tests_properties(screenfx_linux_shaders PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=xcb;LIBGL_ALWAYS_SOFTWARE=1" TIMEOUT 60)
    else()
        add_test(NAME screenfx_linux_shaders COMMAND screenfx_linux_shader_tests)
        set_tests_properties(screenfx_linux_shaders PROPERTIES TIMEOUT 60)
    endif()
endif()
